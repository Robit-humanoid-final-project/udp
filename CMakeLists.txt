cmake_minimum_required(VERSION 3.8)
project(udp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(humanoid_interfaces REQUIRED)
add_executable(udp_node src/udp_node.cpp)
target_compile_features(udp_node PRIVATE cxx_std_17)
ament_target_dependencies(udp_node rclcpp geometry_msgs humanoid_interfaces)
install(TARGETS udp_node DESTINATION lib/${PROJECT_NAME})
ament_package()
